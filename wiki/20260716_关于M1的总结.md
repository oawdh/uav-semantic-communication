## 实体无人机：
from djitellopy import Tello
import cv2
import time
import os

SAVE_IMG_PATH = "./data/capture/"
if not os.path.exists(SAVE_IMG_PATH):
    os.makedirs(SAVE_IMG_PATH)

class DroneController:
    def __init__(self):
        # 去掉 simulator 参数，官方库不支持
        self.drone = Tello()
        self.connected = False
        self.battery = 100

    def connect(self):
        """建立无人机连接，模拟连接失败故障"""
        try:
            self.drone.connect()
            self.connected = True
            self.battery = self.drone.get_battery()
            print(f"连接成功，当前电量：{self.battery}%")
        except Exception as e:
            self.connected = False
            raise Exception(f"【连接失败故障】{str(e)}")

    # M1要求标准API 1：takeoff 起飞
    def takeoff(self):
        if not self.connected:
            raise Exception("未连接无人机，无法起飞")
        if self.battery < 20:
            raise Exception(f"【低电量故障】当前电量{self.battery}%，低于20%禁止起飞")
        try:
            self.drone.takeoff()
            print("执行起飞完成")
            return {"status": "success", "action": "takeoff"}
        except Exception as e:
            raise Exception(f"【指令超时故障】起飞指令无响应：{str(e)}")

    # M1标准API 2：land 降落
    def land(self):
        if not self.connected:
            raise Exception("未连接无人机，无法降落")
        try:
            self.drone.land()
            print("执行降落完成")
            return {"status": "success", "action": "land"}
        except Exception as e:
            raise Exception(f"【指令超时故障】降落指令无响应：{str(e)}")

    # M1标准API3：move 平移 direction: up/down/left/right/forward/back 单位cm
    def move(self, direction: str, distance: int):
        if not self.connected:
            raise Exception("未连接无人机")
        # 模拟障碍物风险：移动距离过大触发保护
        if distance > 100:
            raise Exception("【障碍物风险】移动距离超出安全范围，前方存在障碍物")
        try:
            move_func = getattr(self.drone, f"move_{direction}")
            move_func(distance)
            print(f"平移{direction} {distance}cm 完成")
            return {"status": "success", "action": "move", "dir": direction, "dist": distance}
        except Exception as e:
            raise Exception(f"【指令超时故障】平移指令失败：{str(e)}")

    # M1标准API4：rotate 旋转 angle正数顺时针，负数逆时针
    def rotate(self, angle: int):
        if not self.connected:
            raise Exception("未连接无人机")
        try:
            if angle > 0:
                self.drone.rotate_clockwise(angle)
            else:
                self.drone.rotate_counter_clockwise(abs(angle))
            print(f"旋转{angle}°完成")
            return {"status": "success", "action": "rotate", "angle": angle}
        except Exception as e:
            raise Exception(f"【指令超时故障】旋转指令失败：{str(e)}")

    # M1标准API5：capture_image 拍照保存
    def capture_image(self):
        if not self.connected:
            raise Exception("未连接无人机")
        try:
            frame = self.drone.get_frame_read().frame
            save_name = f"{SAVE_IMG_PATH}img_{int(time.time())}.jpg"
            cv2.imwrite(save_name, frame)
            print(f"拍照完成，图片保存至：{save_name}")
            return {"status": "success", "action": "capture_image", "path": save_name}
        except Exception as e:
            raise Exception(f"【指令超时故障】拍照失败：{str(e)}")

    # M1标准API6：emergency_stop 紧急停机
    def emergency_stop(self):
        try:
            self.drone.emergency()
            self.connected = False
            print("紧急停机触发，无人机锁桨")
            return {"status": "success", "action": "emergency_stop"}
        except Exception as e:
            raise Exception(f"急停指令异常：{str(e)}")

    def close(self):
        """释放连接资源"""
        self.drone.end()
        self.connected = False
